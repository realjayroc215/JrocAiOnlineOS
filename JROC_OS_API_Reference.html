import { loadConfig } from "../core/config"; import { regionsService } from "../regions/regions.service"; import { clustersService } from "../clusters/clusters.service"; import { meshOrchestrator } from "../mesh/mesh-orchestrator.service"; import { metricsService } from "../metrics/metrics.service"; import { logsService } from "../logs/logs.service"; import { eventsService } from "../events/events.service"; import { tracingService } from "../tracing/tracing.service"; import { jobsService } from "../jobs/jobs.service"; export async function runCloudBootstrap() { const config = loadConfig(); console.log("=== JROC OS Cloud Edition Master Bootstrap ==="); console.log("Config loaded:", config); const tenantId = "bootstrap-tenant"; const orgId = "bootstrap-org"; const defaultRegions = ["us-east", "us-west", "eu-central"]; console.log("Bootstrapping regions..."); for (const code of defaultRegions) { await regionsService.ensureRegion(code); } console.log("Bootstrapping initial cluster..."); const cluster = await clustersService.createCluster({ tenantId, orgId, name: "bootstrap-cluster", region: "us-east", plan: "free" }); console.log("Provisioning mesh engines..."); await meshOrchestrator.provisionCluster(cluster); console.log("Seeding metrics..."); await metricsService.ingest({ tenantId, clusterId: cluster.id, name: "cpu_usage", value: 0.1 }); await metricsService.ingest({ tenantId, clusterId: cluster.id, name: "req_per_sec", value: 10 }); console.log("Seeding logs..."); await logsService.ingest({ tenantId, clusterId: cluster.id, level: "info", message: "Bootstrap cluster created", context: { phase: "bootstrap" } }); console.log("Seeding events..."); await eventsService.ingest({ tenantId, clusterId: cluster.id, type: "bootstrap.cluster.created", message: "Bootstrap cluster created successfully" }); console.log("Seeding tracing..."); const start = new Date().toISOString(); const end = new Date(Date.now() + 500).toISOString(); await tracingService.ingest({ tenantId, clusterId: cluster.id, traceId: "bootstrap-trace", name: "bootstrap-sequence", startTime: start, endTime: end, attributes: { clusterId: cluster.id, tenantId } }); console.log("Enqueuing bootstrap job..."); await jobsService.enqueue({ tenantId, clusterId: cluster.id, type: "bootstrap.verify", payload: { clusterId: cluster.id } }); console.log("=== Cloud Bootstrap COMPLETE ==="); return { config, cluster }; } // keep optional standalone entry if (require.main === module) { runCloudBootstrap().catch(err => { console.error("Bootstrap failed:", err); process.exit(1); }); } \


#!/usr/bin/env bash
set -e

echo "🔥 JROC AI OS – MEGA LIFECYCLE"
echo ""

# 1. Bootstrap all subsystems
echo "🚀 [1/6] jroc bootstrap"
jroc bootstrap

# 2. Start full ecosystem
echo ""
echo "⚙️ [2/6] jroc run"
jroc run &

# give services a moment to come up
sleep 5

# 3. Switch to Studio OS mode
echo ""
echo "🎛️ [3/6] jroc mode studio-os"
jroc mode studio-os

# 4. Deploy to cloud
echo ""
echo "☁️ [4/6] jroc deploy cloud"
jroc deploy cloud

# 5. Run operational tests
echo ""
echo "🧪 [5/6] jroc test ops"
jroc test ops

# 6. Show system status
echo ""
echo "📊 [6/6] jroc status"
jroc status

echo ""
echo "✅ JROC AI OS Mega Lifecycle complete."
